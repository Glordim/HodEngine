file(GLOB_RECURSE Srcs
	"${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/*.hpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/*.inl"
	"${CMAKE_CURRENT_SOURCE_DIR}/*.h"
)

file(GLOB_RECURSE Includes
	"${CMAKE_CURRENT_SOURCE_DIR}/*.hpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/*.inl"
	"${CMAKE_CURRENT_SOURCE_DIR}/*.h"
)

assign_source_group(${Srcs}) # for VS

if(HOD_PROJECT_BUILD_TYPE STREQUAL "Editor" OR ANDROID)
	add_library([[PROJECT_NAME]] SHARED ${Srcs} ${Includes})
	add_library([[PROJECT_NAME]]::[[PROJECT_NAME]] ALIAS [[PROJECT_NAME]])
else()
	add_executable([[PROJECT_NAME]] ${Srcs} ${Includes})
endif()

install(TARGETS [[PROJECT_NAME]]
	RUNTIME DESTINATION .
	LIBRARY DESTINATION .
	ARCHIVE DESTINATION .
)

target_include_directories([[PROJECT_NAME]] PUBLIC
	"${CMAKE_CURRENT_SOURCE_DIR}/.."
)

target_link_libraries([[PROJECT_NAME]]
	HodEngine::Core
	HodEngine::Math
	HodEngine::Window
	HodEngine::Renderer
	HodEngine::Input
	HodEngine::GameSystems
	HodEngine::Game
	HodEngine::ImGui
	HodEngine::Application
)

hod_read_engine_modules("${CMAKE_SOURCE_DIR}/Cache/Generated/HodModules.txt" HOD_ENGINE_MODULES)
hod_link_engine_modules([[PROJECT_NAME]] ${HOD_ENGINE_MODULES})

if(HOD_PROJECT_BUILD_TYPE STREQUAL "Application")
	hod_add_engine_modules_entry_points([[PROJECT_NAME]] ${HOD_ENGINE_MODULES} [[PROJECT_NAME]]) # The project's runtime module last
	if(ANDROID)
		target_link_options([[PROJECT_NAME]] PRIVATE
			"-Wl,--undefined=Java_com_google_androidgamesdk_GameActivity_initializeNativeCode"
		)
	endif()
	target_link_libraries([[PROJECT_NAME]] HodEngine::Main)
	hod_deploy_runtime_files([[PROJECT_NAME]])
endif()

target_compile_definitions([[PROJECT_NAME]]
	PUBLIC ${HOD_PLATFORM_DEFINE}
	PRIVATE [[PROJECT_EXPORT]]
)
